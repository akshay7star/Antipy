# Dijkstra's Algorithm: GPS for Graphs

When a graph has **weighted edges** (distances, costs, latency), BFS no longer finds the shortest path. Dijkstra's algorithm solves this using a **greedy strategy** with a priority queue: always expand the cheapest known path next.

This is the algorithm powering GPS navigation, network routing (OSPF), and game AI pathfinding.

## The Algorithm

```python
import heapq

def dijkstra(graph, start):
    # graph = {node: [(neighbor, weight), ...]}
    distances = {node: float('inf') for node in graph}
    distances[start] = 0
    priority_queue = [(0, start)]   # (distance, node)
    previous = {}

    while priority_queue:
        curr_dist, curr_node = heapq.heappop(priority_queue)

        # Skip if we already found a shorter path to this node
        if curr_dist > distances[curr_node]:
            continue

        for neighbor, weight in graph[curr_node]:
            new_dist = curr_dist + weight

            if new_dist < distances[neighbor]:
                distances[neighbor] = new_dist
                previous[neighbor] = curr_node
                heapq.heappush(priority_queue, (new_dist, neighbor))

    return distances, previous
```

## Step-by-Step Trace

```
Graph: A--(2)--C, A--(4)--B, C--(1)--B, B--(3)--D, C--(5)--D
Start: A

Initial: dist={A:0, B:∞, C:∞, D:∞}, PQ=[(0,A)]

Step 1: Pop (0,A). Relax edges:
  A→C: 0+2=2 < ∞ → dist[C]=2, push (2,C)
  A→B: 0+4=4 < ∞ → dist[B]=4, push (4,B)

Step 2: Pop (2,C). Relax edges:
  C→B: 2+1=3 < 4 → dist[B]=3, push (3,B)
  C→D: 2+5=7 < ∞ → dist[D]=7, push (7,D)

Step 3: Pop (3,B). Relax edges:
  B→D: 3+3=6 < 7 → dist[D]=6, push (6,D)

Step 4: Pop (4,B). curr_dist=4 > dist[B]=3 → SKIP!

Step 5: Pop (6,D). D reached!

Final: {A:0, B:3, C:2, D:6}
```

## Path Reconstruction

```python
def reconstruct_path(previous, start, end):
    path = []
    current = end
    while current != start:
        path.append(current)
        current = previous[current]
    path.append(start)
    return list(reversed(path))
```

<interactive-code>
import heapq
from collections import defaultdict

def dijkstra(graph, start):
    distances = defaultdict(lambda: float('inf'))
    distances[start] = 0
    previous = {}
    pq = [(0, start)]

    while pq:
        dist, node = heapq.heappop(pq)
        if dist > distances[node]:
            continue  # Stale entry
        for neighbor, weight in graph[node]:
            new_dist = dist + weight
            if new_dist < distances[neighbor]:
                distances[neighbor] = new_dist
                previous[neighbor] = node
                heapq.heappush(pq, (new_dist, neighbor))

    return dict(distances), previous

def get_path(previous, start, end):
    if end not in previous and end != start:
        return None
    path = []
    curr = end
    while curr != start:
        path.append(curr)
        curr = previous.get(curr)
        if curr is None:
            return None
    path.append(start)
    return list(reversed(path))

# --- Test 1: Simple weighted graph ---
print("=== Dijkstra on Simple Graph ===")
graph = {
    'A': [('B', 4), ('C', 2)],
    'B': [('D', 3)],
    'C': [('B', 1), ('D', 5)],
    'D': [],
}
distances, previous = dijkstra(graph, 'A')
print(f"Shortest distances from A: {dict(distances)}")
for target in ['B', 'C', 'D']:
    path = get_path(previous, 'A', target)
    cost = distances[target]
    print(f"  A → {target}: {' → '.join(path)} (cost={cost})")

# --- Test 2: Network with multiple paths ---
print("\n=== Network Routing Graph ===")
network = {
    's': [('a', 10), ('c', 3)],
    'a': [('b', 2)],
    'b': [('t', 1)],
    'c': [('a', 4), ('d', 8), ('t', 2)],
    'd': [('b', 5), ('t', 7)],
    't': [],
}
dist, prev = dijkstra(network, 's')
path_to_t = get_path(prev, 's', 't')
print(f"Shortest s → t: {' → '.join(path_to_t)} (cost={dist['t']})")
print(f"All distances from s: {dist}")

# --- Test 3: Grid with costs ---
print("\n=== Grid Dijkstra (Weighted Grid) ===")
grid_costs = [
    [1, 3, 1, 1],
    [1, 5, 1, 1],
    [4, 2, 1, 1],
    [1, 1, 1, 1],
]
rows, cols = len(grid_costs), len(grid_costs[0])
dist_grid = [[float('inf')] * cols for _ in range(rows)]
dist_grid[0][0] = grid_costs[0][0]
pq = [(grid_costs[0][0], 0, 0)]
dirs = [(0,1),(0,-1),(1,0),(-1,0)]

while pq:
    d, r, c = heapq.heappop(pq)
    if d > dist_grid[r][c]:
        continue
    for dr, dc in dirs:
        nr, nc = r+dr, c+dc
        if 0 <= nr < rows and 0 <= nc < cols:
            new_d = d + grid_costs[nr][nc]
            if new_d < dist_grid[nr][nc]:
                dist_grid[nr][nc] = new_d
                heapq.heappush(pq, (new_d, nr, nc))

print(f"Min cost path from (0,0) to (3,3): {dist_grid[3][3]}")
print("Distance grid:")
for row in dist_grid:
    print("  " + "  ".join(f"{v:2}" for v in row))

## Complexity Analysis

| Implementation | Time | Space |
|---|---|---|
| Binary heap (heapq) | O((V + E) log V) | O(V + E) |
| Fibonacci heap | O(E + V log V) | O(V + E) |
| Simple array (no heap) | O(V²) | O(V) — good for dense graphs |

> **Why `if curr_dist > distances[curr_node]: continue`?** The priority queue can contain multiple entries for the same node (from successive relaxations). When we pop a stale entry with higher cost than what we already know, we skip it. This "lazy deletion" avoids rebuilding the heap.

## Dijkstra's Limitations

```python
# Dijkstra FAILS with negative edge weights!
# A negative edge can make a "longer" path cheaper,
# but Dijkstra assumes once a node is popped, its distance is final.

# Example where Dijkstra gives wrong answer:
wrong_graph = {
    'A': [('B', 1), ('C', 4)],
    'B': [('C', -3)],   # Negative weight!
    'C': [],
}
# Dijkstra: A→C = 4. But actual shortest: A→B→C = 1+(-3) = -2!

# Solution: Use Bellman-Ford for graphs with negative weights
# (handles negative edges, detects negative cycles)
```

> **Pro Tip:** In interviews, Dijkstra problems often disguise themselves: "minimum cost path," "network delay time," "cheapest flights within k stops." The pattern is always: weighted graph + shortest path = Dijkstra with a heap. When you see "k stops," use a modified Dijkstra with state (node, stops_remaining).

> **Common Mistake:** Forgetting to skip stale priority queue entries. Without `if curr_dist > distances[node]: continue`, you'll reprocess nodes with old distances, potentially overwriting shorter paths and degrading to O(V·E) in the worst case.
